from pathlib import Path

carousel = Path("src/features/media/MediaCarousel.jsx")
css = Path("src/features/media/media.css")

if not carousel.exists():
    raise SystemExit("Missing src/features/media/MediaCarousel.jsx")

if not css.exists():
    raise SystemExit("Missing src/features/media/media.css")

carousel_backup = carousel.with_name(
    "MediaCarousel.jsx.before-saved-orientation-full-fix"
)
css_backup = css.with_name(
    "media.css.before-saved-orientation-full-fix"
)

carousel_backup.write_text(carousel.read_text())
css_backup.write_text(css.read_text())

carousel.write_text("""import {
  ChevronLeft,
  ChevronRight,
  Image as ImageIcon
} from 'lucide-react'
import {useEffect, useMemo, useState} from 'react'
import {isDisplayablePhotoSource} from './imagePersistence'

const clamp = (value, minimum, maximum) => {
  const numeric = Number(value)

  if (!Number.isFinite(numeric)) {
    return minimum
  }

  return Math.min(
    maximum,
    Math.max(minimum, numeric)
  )
}

const getSource = photo =>
  photo?.signed_url ||
  photo?.url ||
  photo?.data_url ||
  photo?.preview

export default function MediaCarousel({
  photos = [],
  fallbackClass = 'media-fallback',
  positionX = 50,
  positionY = 50,
  featuredIndex = 0
}) {
  const valid = useMemo(
    () =>
      photos.filter(photo =>
        isDisplayablePhotoSource(getSource(photo))
      ),
    [photos]
  )

  const safeFeaturedIndex = Math.min(
    Math.max(0, Number(featuredIndex || 0)),
    Math.max(0, valid.length - 1)
  )

  const [index, setIndex] = useState(
    safeFeaturedIndex
  )

  useEffect(() => {
    setIndex(safeFeaturedIndex)
  }, [safeFeaturedIndex, valid.length])

  const safeIndex = Math.min(
    Math.max(0, index),
    Math.max(0, valid.length - 1)
  )

  const photo = valid[safeIndex]
  const source = getSource(photo)

  if (!source) {
    return (
      <div
        className={`media-carousel ${fallbackClass}`}
      >
        <ImageIcon size={34} />
      </div>
    )
  }

  const x = clamp(
    photo?.position_x ?? positionX ?? 50,
    0,
    100
  )

  const y = clamp(
    photo?.position_y ?? positionY ?? 50,
    0,
    100
  )

  const scale = clamp(
    photo?.crop_scale ?? 1,
    1,
    3
  )

  const imageStyle = {
    objectPosition: `${x}% ${y}%`,
    transform: `scale(${scale})`,
    transformOrigin: `${x}% ${y}%`
  }

  return (
    <div
      className={`media-carousel ${fallbackClass}`}
    >
      <img
        src={source}
        alt=""
        draggable="false"
        style={imageStyle}
        onError={event => {
          event.currentTarget.style.display = 'none'
        }}
      />

      {safeIndex === safeFeaturedIndex && (
        <span className="media-featured-badge">
          ★ Featured
        </span>
      )}

      {valid.length > 1 && (
        <>
          <button
            type="button"
            className="media-prev"
            aria-label="Previous photograph"
            onClick={event => {
              event.stopPropagation()

              setIndex(
                (
                  safeIndex -
                  1 +
                  valid.length
                ) % valid.length
              )
            }}
          >
            <ChevronLeft />
          </button>

          <button
            type="button"
            className="media-next"
            aria-label="Next photograph"
            onClick={event => {
              event.stopPropagation()

              setIndex(
                (safeIndex + 1) % valid.length
              )
            }}
          >
            <ChevronRight />
          </button>

          <span className="media-count">
            {safeIndex + 1}/{valid.length}
          </span>

          <div className="media-dots">
            {valid.map((item, photoIndex) => (
              <i
                className={
                  photoIndex === safeIndex
                    ? 'active'
                    : ''
                }
                key={
                  item.id ||
                  item.storage_path ||
                  item.path ||
                  photoIndex
                }
              />
            ))}
          </div>
        </>
      )}
    </div>
  )
}
""")

css_text = css.read_text()
marker = "/* Saved activity photo framing */"

if marker in css_text:
    css_text = css_text[:css_text.index(marker)].rstrip() + "\n"

css_text += r'''

/* Saved activity photo framing */
.media-carousel {
  position: relative;
  overflow: hidden !important;
}

.media-carousel > img {
  display: block;
  width: 100% !important;
  height: 100% !important;
  min-width: 100% !important;
  min-height: 100% !important;
  max-width: none !important;
  object-fit: cover !important;
  transition:
    transform .2s ease,
    object-position .2s ease;
  will-change: transform, object-position;
}
'''

css.write_text(css_text)

print("[OK] Replaced MediaCarousel with valid JSX")

print("[OK] Each slider photo uses its own saved framing")
print("[OK] Featured photograph remains selected")
print("[OK] Carousel backup:", carousel_backup)

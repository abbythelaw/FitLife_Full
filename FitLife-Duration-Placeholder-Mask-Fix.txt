from pathlib import Path

path = Path("src/features/sports/SportsEntryForm.jsx")
text = path.read_text()

backup = path.with_name(
    "SportsEntryForm.jsx.before-duration-mask-fix"
)
backup.write_text(text)

old_state = """  const [durationInput, setDurationInput] = useState(() =>
    secondsToDuration(
      initial.duration_seconds ??
      Number(initial.duration_minutes || 45) * 60
    )
  )

  const durationSeconds = useMemo(
    () => durationToSeconds(durationInput),
    [durationInput]
  )

  const duration = durationSeconds / 60"""

new_state = """  const [durationDigitsInput, setDurationDigitsInput] = useState(() => {
    const existingSeconds =
      Number(initial.duration_seconds) ||
      Number(initial.duration_minutes || 0) * 60

    if (!existingSeconds) return ''

    return secondsToDuration(existingSeconds).replace(/\\D/g, '')
  })

  const durationInput = durationDigitsInput
    ? formatDurationDigits(durationDigitsInput)
    : ''

  const durationSeconds = useMemo(
    () => durationDigitsInput
      ? durationToSeconds(durationDigitsInput)
      : 0,
    [durationDigitsInput]
  )

  const duration = durationSeconds / 60"""

if old_state not in text:
    raise SystemExit(
        "Current duration state block not found. No changes made."
    )

text = text.replace(old_state, new_state, 1)

old_input = """          <input
            className="sports-duration-input"
            type="text"
            inputMode="numeric"
            autoComplete="off"
            maxLength="8"
            value={durationInput}
            placeholder="00:00:00"
            aria-label="Duration in hours, minutes and seconds"
            onFocus={event => event.currentTarget.select()}
            onChange={event => {
              changed(
                setDurationInput,
                formatDurationDigits(event.target.value)
              )
            }}
            onKeyDown={event => {
              if (event.key !== 'Backspace') return

              event.preventDefault()

              const digits = durationDigits(durationInput)
                .replace(/^0+/, '')
                .slice(0, -1)

              changed(
                setDurationInput,
                formatDurationDigits(digits)
              )
            }}
            required
          />
          <small className="sports-duration-help">
            Type numbers only. For example, 13000 becomes 01:30:00.
          </small>"""

new_input = """          <input
            className="sports-duration-input"
            type="text"
            inputMode="numeric"
            autoComplete="off"
            maxLength="8"
            value={durationInput}
            placeholder="HH:MM:SS"
            aria-label="Duration in hours, minutes and seconds"
            onChange={event => {
              const digits = event.target.value
                .replace(/\\D/g, '')
                .slice(-6)

              changed(setDurationDigitsInput, digits)
            }}
            onKeyDown={event => {
              if (event.key === 'Backspace') {
                event.preventDefault()

                changed(
                  setDurationDigitsInput,
                  durationDigitsInput.slice(0, -1)
                )
              }

              if (event.key === 'Delete') {
                event.preventDefault()
                changed(setDurationDigitsInput, '')
              }
            }}
            required
          />
          <small className="sports-duration-help">
            Type numbers only. Example: 13000 becomes 01:30:00.
          </small>"""

if old_input not in text:
    raise SystemExit(
        "Current duration input block not found. No changes made."
    )

text = text.replace(old_input, new_input, 1)
path.write_text(text)

print("[OK] Duration starts empty")
print("[OK] Grey HH:MM:SS placeholder added")
print("[OK] Placeholder disappears when typing")
print("[OK] Numeric entry formats from the right")
print("[OK] Backspace removes one typed digit")
print("[OK] Delete clears the duration")
print("[OK] Existing activities load saved duration")
print("[OK] Backup:", backup)
PY
